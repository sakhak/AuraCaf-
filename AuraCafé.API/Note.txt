=============== Generate a random JWT key using PowerShell ===============
$bytes = [byte[]]::new(48)
[System.Security.Cryptography.RandomNumberGenerator]::Fill($bytes)
$key = [Convert]::ToBase64String($bytes)

dotnet user-secrets set "Jwt:Key" $key --project AuraCafé.API
